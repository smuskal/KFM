# Pin EXACTLY. A joblib-pickled random forest is a pickled object graph: a
# different scikit-learn either refuses to load it or loads it and scores
# differently. These are the versions the models were fitted with.
scikit-learn==1.7.2
numpy==2.2.6
joblib==1.5.3

# rdkit is deliberately NOT pinned to one release, and it is the only one here
# that is not. It never touches the pickled forest -- it computes the Morgan
# count fingerprint and the descriptors, which are stable across releases.
# Verified: rdkit 2025.03.6 and 2025.09.5 return predictions identical to ten
# decimal places on the same bundle.
#
# Pinning it broke installation outright on x86_64 macOS, where rdkit stopped
# publishing wheels after 2025.9.3. That includes Apple Silicon machines running
# an Intel Python under Rosetta, which is easy not to notice. Check yours with:
#   python -c "import platform; print(platform.machine())"
# arm64 is native, x86_64 is Intel or Rosetta. Both work with the floor below.
rdkit>=2024.3.1

# Do NOT mix conda-forge rdkit with pip numpy/scipy in one environment; it breaks
# scipy's extension modules. Pick one packaging system.

# Unpinned, and here for a reason: environment.yml installs pytest, so the conda
# path could run the suite and the venv fallback could not. Anyone who installs
# without conda is exactly the person who most needs to verify the install, and
# they hit "No module named pytest" at that moment. Kept unpinned to match
# environment.yml -- tests/test_kfm.py compares only the constrained pins, so an
# unpinned name in one file and not the other would not have been caught.
pytest
